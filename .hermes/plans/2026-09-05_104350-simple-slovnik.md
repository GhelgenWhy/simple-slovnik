# Simple Slovník — словарь чешский ↔ укр/рус

> **For Hermes:** десктоп-приложение с локальной SQLite-базой слов, CRUD + быстрый поиск + чешский spellcheck с подсказками.

**Goal:** Персональный офлайн-словарь с мгновенным поиском, возможностью добавлять/редактировать/удалять записи, и проверкой написания чешских слов с предложением ближайших правильных вариантов.

**Architecture:** Python 3.11+ десктоп-приложение на PySide6 (Qt), SQLite для хранения слов и тегов, `hunspell` (через `pyhunspell`) для проверки чешской грамматики с нативным словарём OpenOffice `.dic`/`.aff`. Многооконный интерфейс: главное окно (поиск + список) + диалоги добавления/редактирования. Импорт/экспорт CSV для миграций.

**Tech Stack:**
- Python 3.11+
- PySide6 (Qt для Python, LGPL)
- SQLite3 (встроено)
- `pyhunspell` (нативная проверка чешского) + словарь `cs_CZ` из OpenOffice
- pytest для тестов
- `python-Levenshtein` (опционально, для нечёткого поиска)

## Модель данных

```sql
CREATE TABLE words (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    cs          TEXT    NOT NULL,              -- чешское слово
    uk          TEXT    DEFAULT '',            -- украинский перевод
    ru          TEXT    DEFAULT '',            -- русский перевод
    pos         TEXT    DEFAULT '',            -- часть речи (тег): noun/verb/adj/adv/...
    note        TEXT    DEFAULT '',            -- заметка пользователя
    created_at  TEXT    NOT NULL DEFAULT (datetime('now')),
    updated_at  TEXT    NOT NULL DEFAULT (datetime('now')),
    UNIQUE(cs COLLATE NOCASE)
);

CREATE INDEX idx_words_cs ON words(cs COLLATE NOCASE);
CREATE INDEX idx_words_uk ON words(uk COLLATE NOCASE);
CREATE INDEX idx_words_ru ON words(ru COLLATE NOCASE);

CREATE VIRTUAL TABLE words_fts USING fts5(
    cs, uk, ru, note,
    content='words',
    content_rowid='id',
    tokenize="unicode61"
);

-- триггеры для синхронизации FTS
```

**Поиск:** по `LIKE` с `COLLATE NOCASE` для префиксного, плюс FTS5 для полнотекстового.

```python
import sqlite3
from pathlib import Path
from typing import Optional, List
from .models import Word

SCHEMA = """
CREATE TABLE IF NOT EXISTS words (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    cs TEXT NOT NULL,
    uk TEXT DEFAULT '',
    ru TEXT DEFAULT '',
    pos TEXT DEFAULT '',
    note TEXT DEFAULT '',
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now')),
    UNIQUE(cs COLLATE NOCASE)
);
CREATE INDEX IF NOT EXISTS idx_words_cs ON words(cs COLLATE NOCASE);
CREATE INDEX IF NOT EXISTS idx_words_uk ON words(uk COLLATE NOCASE);
CREATE INDEX IF NOT EXISTS idx_words_ru ON words(ru COLLATE NOCASE);
"""

def _conn(db_path: str) -> sqlite3.Connection:
    c = sqlite3.connect(db_path)
    c.row_factory = sqlite3.Row
    c.execute("PRAGMA foreign_keys = ON")
    return c

def init_db(db_path: str) -> None:
    """Создаёт файл БД и таблицы, если их нет."""
    Path(db_path).parent.mkdir(parents=True, exist_ok=True)
    with _conn(db_path) as c:
        c.executescript(SCHEMA)
        c.commit()

def _row_to_word(r: sqlite3.Row) -> Word:
    return Word(id=r["id"], cs=r["cs"], uk=r["uk"], ru=r["ru"],
                pos=r["pos"], note=r["note"],
                created_at=r["created_at"], updated_at=r["updated_at"])

def add_word(db_path: str, cs: str, uk: str = "", ru: str = "",
             pos: str = "", note: str = "") -> int:
    with _conn(db_path) as c:
        cur = c.execute(
            "INSERT INTO words (cs, uk, ru, pos, note) VALUES (?, ?, ?, ?, ?)",
            (cs, uk, ru, pos, note),
        )
        c.commit()
        return cur.lastrowid

def get_word(db_path: str, word_id: int) -> Optional[Word]:
    with _conn(db_path) as c:
        r = c.execute("SELECT * FROM words WHERE id = ?", (word_id,)).fetchone()
        return _row_to_word(r) if r else None

def list_words(db_path: str, limit: int = 500) -> List[Word]:
    with _conn(db_path) as c:
        rs = c.execute("SELECT * FROM words ORDER BY cs COLLATE NOCASE LIMIT ?", (limit,)).fetchall()
        return [_row_to_word(r) for r in rs]

def update_word(db_path: str, word_id: int, **fields) -> None:
    if not fields:
        return
    sets = ", ".join(f"{k} = ?" for k in fields)
    vals = list(fields.values()) + [word_id]
    with _conn(db_path) as c:
        c.execute(f"UPDATE words SET {sets}, updated_at = datetime('now') WHERE id = ?", vals)
        c.commit()

def delete_word(db_path: str, word_id: int) -> None:
    with _conn(db_path) as c:
        c.execute("DELETE FROM words WHERE id = ?", (word_id,))
        c.commit()

def search_words(db_path: str, query: str, limit: int = 100) -> List[Word]:
    """Поиск: сначала точное/префиксное по cs, uk, ru, потом LIKE fallback."""
    if not query.strip():
        return list_words(db_path, limit)
    pat = f"%{query}%"
    with _conn(db_path) as c:
        rs = c.execute(
            """SELECT * FROM words
               WHERE cs LIKE ? COLLATE NOCASE
                  OR uk LIKE ? COLLATE NOCASE
                  OR ru LIKE ? COLLATE NOCASE
               ORDER BY
                   CASE WHEN cs LIKE ? COLLATE NOCASE THEN 0 ELSE 1 END,
                   cs COLLATE NOCASE
               LIMIT ?""",
            (pat, pat, pat, f"{query}%", limit),
        ).fetchall()
        return [_row_to_word(r) for r in rs]
```

# src/slovnik/import_export.py
import csv
from typing import Iterable
from .db import init_db, add_word

FIELDS = ["cs", "uk", "ru", "pos", "note"]

def export_csv(db_path: str, csv_path: str) -> int:
    from .db import list_words
    words = list_words(db_path, limit=100_000)
    with open(csv_path, "w", encoding="utf-8", newline="") as f:
        w = csv.DictWriter(f, fieldnames=FIELDS)
        w.writeheader()
        for word in words:
            w.writerow({k: getattr(word, k) for k in FIELDS})
    return len(words)

def import_csv(db_path: str, csv_path: str, skip_existing: bool = True) -> int:
    """Возвращает количество импортированных строк."""
    init_db(db_path)
    count = 0
    with open(csv_path, "r", encoding="utf-8", newline="") as f:
        r = csv.DictReader(f)
        for row in r:
            cs = (row.get("cs") or "").strip()
            if not cs:
                continue
            try:
                add_word(
                    db_path,
                    cs=cs,
                    uk=(row.get("uk") or "").strip(),
                    ru=(row.get("ru") or "").strip(),
                    pos=(row.get("pos") or "").strip(),
                    note=(row.get("note") or "").strip(),
                )
                count += 1
            except Exception:
                # UNIQUE конфликт — пропускаем если skip_existing
                if not skip_existing:
                    raise
    return count
```

# src/slovnik/spellcheck.py
from pathlib import Path
from typing import List, Optional
import re

try:
    import hunspell
    _HAS_HUNSPELL = True
except ImportError:
    _HAS_HUNSPELL = False

DICT_DIR = Path(__file__).resolve().parent.parent.parent / "dicts" / "cs_CZ"

class SpellChecker:
    def __init__(self, dict_dir: Optional[Path] = None):
        self.dict_dir = Path(dict_dir) if dict_dir else DICT_DIR
        self._h = None
        if _HAS_HUNSPELL and (self.dict_dir / "cs_CZ.aff").exists():
            self._h = hunspell.HunSpell(
                str(self.dict_dir / "cs_CZ.dic"),
                str(self.dict_dir / "cs_CZ.aff"),
            )

    def is_correct(self, word: str) -> bool:
        word = word.strip()
        if not word:
            return True
        if self._h is None:
            return self._fallback_check(word)
        return self._h.spell(word)

    def suggest(self, word: str, limit: int = 8) -> List[str]:
        word = word.strip()
        if not word:
            return []
        if self._h is None:
            return self._fallback_suggest(word, limit)
        return self._h.suggest(word)[:limit]

    def _fallback_check(self, word: str) -> bool:
        """Если hunspell недоступен — всегда True (мягкий режим)."""
        return True

    def _fallback_suggest(self, word: str, limit: int) -> List[str]:
        return []
```

from pathlib import Path
from typing import Optional
from PySide6.QtCore import Qt, QTimer
from PySide6.QtGui import QAction, QKeySequence
from PySide6.QtWidgets import (
    QMainWindow, QWidget, QVBoxLayout, QHBoxLayout,
    QLineEdit, QListWidget, QListWidgetItem, QPushButton,
    QLabel, QStatusBar, QMessageBox, QFileDialog,
)
from ..db import init_db, list_words, search_words, delete_word
from ..models import Word
from ..import_export import export_csv, import_csv

class MainWindow(QMainWindow):
    def __init__(self, db_path: Path):
        super().__init__()
        self.db_path = str(db_path)
        init_db(self.db_path)
        self.setWindowTitle("Slovník — cs ↔ uk/ru")
        self.resize(900, 600)

        # Виджеты
        self.search = QLineEdit()
        self.search.setPlaceholderText("🔍 Пошук: cs / uk / ru…  (Ctrl+F)")
        self.search.textChanged.connect(self._debounced_refresh)

        self.list_widget = QListWidget()
        self.list_widget.itemDoubleClicked.connect(self._on_edit)

        self.btn_add = QPushButton("➕ Додати (Ctrl+N)")
        self.btn_edit = QPushButton("✏️ Редагувати (Enter)")
        self.btn_del = QPushButton("🗑️ Видалити (Del)")
        self.btn_export = QPushButton("📤 Експорт CSV")
        self.btn_import = QPushButton("📥 Імпорт CSV")
        self.btn_check = QPushButton("✅ Перевірити")

        # Layout
        central_widget = QWidget()
        layout = QVBoxLayout(central_widget)
        top = QHBoxLayout()
        top.addWidget(self.search)
        top.addWidget(self.btn_check)
        layout.addLayout(top)
        layout.addWidget(self.list_widget)

        bottom = QHBoxLayout()
        bottom.addWidget(self.btn_add)
        bottom.addWidget(self.btn_edit)
        bottom.addWidget(self.btn_del)
        bottom.addStretch()
        bottom.addWidget(self.btn_import)
        bottom.addWidget(self.btn_export)
        layout.addLayout(bottom)
        self.setCentralWidget(central_widget)

        self.status = QStatusBar()
        self.setStatusBar(self.status)

        # Сигналы
        self.btn_add.clicked.connect(self._on_add)
        self.btn_edit.clicked.connect(self._on_edit)
        self.btn_del.clicked.connect(self._on_delete)
        self.btn_export.clicked.connect(self._on_export)
        self.btn_import.clicked.connect(self._on_import)
        self.btn_check.clicked.connect(self._on_check_selected)

        # Хоткеи
        QAction("Find", self, shortcut=QKeySequence("Ctrl+F"), triggered=self.search.setFocus)
        QAction("Add", self, shortcut=QKeySequence("Ctrl+N"), triggered=self._on_add)
        QAction("Delete", self, shortcut="Delete", triggered=self._on_delete)

        # Debounce поиска
        self._search_timer = QTimer(self)
        self._search_timer.setSingleShot(True)
        self._search_timer.setInterval(150)
        self._search_timer.timeout.connect(self._refresh_list)

        self._refresh_list()

    def _debounced_refresh(self):
        self._search_timer.start()

    def _refresh_list(self):
        self.list_widget.clear()
        q = self.search.text().strip()
        words = search_words(self.db_path, q) if q else list_words(self.db_path)
        for w in words:
            item = QListWidgetItem(self._format_word(w))
            item.setData(Qt.UserRole, w.id)
            self.list_widget.addItem(item)
        self.status.showMessage(f"Знайдено: {self.list_widget.count()}")

    def _format_word(self, w: Word) -> str:
        parts = [f"🇨🇿 {w.cs}"]
        if w.pos:
            parts.append(f"[{w.pos}]")
        if w.uk:
            parts.append(f"🇺🇦 {w.uk}")
        if w.ru:
            parts.append(f"🇷🇺 {w.ru}")
        if w.note:
            parts.append(f"— {w.note}")
        return "    ".join(parts)

    def _selected_id(self) -> Optional[int]:
        item = self.list_widget.currentItem()
        return item.data(Qt.UserRole) if item else None

    def _on_add(self):
        from .word_dialog import WordDialog
        dlg = WordDialog(self, self.db_path, word_id=None)
        if dlg.exec():
            self._refresh_list()

    def _on_edit(self):
        wid = self._selected_id()
        if wid is None:
            QMessageBox.information(self, "—", "Виберіть слово для редагування.")
            return
        from .word_dialog import WordDialog
        dlg = WordDialog(self, self.db_path, word_id=wid)
        if dlg.exec():
            self._refresh_list()

    def _on_delete(self):
        wid = self._selected_id()
        if wid is None:
            return
        ans = QMessageBox.question(self, "Видалити?", "Точно видалити це слово?")
        if ans == QMessageBox.Yes:
            delete_word(self.db_path, wid)
            self._refresh_list()

    def _on_export(self):
        path, _ = QFileDialog.getSaveFileName(self, "Експорт CSV", "slovnik.csv", "CSV (*.csv)")
        if path:
            n = export_csv(self.db_path, path)
            self.status.showMessage(f"Експортовано {n} слів")

    def _on_import(self):
        path, _ = QFileDialog.getOpenFileName(self, "Імпорт CSV", "", "CSV (*.csv)")
        if path:
            n = import_csv(self.db_path, path)
            self.status.showMessage(f"Імпортовано {n} слів")
            self._refresh_list()

    def _on_check_selected(self):
        from ..spellcheck import SpellChecker
        wid = self._selected_id()
        if wid is None:

from PySide6.QtWidgets import (
    QDialog, QFormLayout, QLineEdit, QComboBox, QDialogButtonBox,

from ..db import get_word, update_word, add_word
from ..spellcheck import SpellChecker

POS_OPTIONS = ["", "noun", "verb", "adj", "adv", "pron", "prep", "conj", "interj", "phrase", "num"]

class WordDialog(QDialog):
    def __init__(self, parent, db_path: str, word_id=None):
        super().__init__(parent)
        self.db_path = db_path
        self.word_id = word_id
        self.setWindowTitle("Редагувати слово" if word_id else "Додати слово")
        self.resize(420, 280)

        self.cs = QLineEdit()
        self.uk = QLineEdit()
        self.ru = QLineEdit()
        self.pos = QComboBox()
        self.pos.addItems(POS_OPTIONS)
        self.pos.setEditable(True)
        self.note = QLineEdit()

        form = QFormLayout()
        form.addRow("🇨🇿 Česky *:", self.cs)
        form.addRow("🇺🇦 Українська:", self.uk)
        form.addRow("🇷🇺 Русский:", self.ru)
        form.addRow("POS:", self.pos)
        form.addRow("Нотатка:", self.note)

        self.check_btn = QPushButton("🔍 Перевірити cs")

        check_row = QHBoxLayout()
        check_row.addWidget(self.check_btn)
        check_row.addStretch()

        self.buttons = QDialogButtonBox(QDialogButtonBox.Save | QDialogButtonBox.Cancel)
        self.buttons.accepted.connect(self._save)
        self.buttons.rejected.connect(self.reject)

        layout = QVBoxLayout(self)
        layout.addLayout(form)
        layout.addLayout(check_row)
        layout.addWidget(self.buttons)

        if word_id:
            w = get_word(db_path, word_id)
            if w:
                self.cs.setText(w.cs)
                self.uk.setText(w.uk)
                self.ru.setText(w.ru)
                idx = self.pos.findText(w.pos)
                if idx >= 0:
                    self.pos.setCurrentIndex(idx)
                else:
                    self.pos.setEditText(w.pos)
                self.note.setText(w.note)

    def _check_cs(self):
        word = self.cs.text().strip()
        if not word:
            return
        sc = SpellChecker()
        if sc.is_correct(word):
            QMessageBox.information(self, "✅ OK", f"«{word}» — правильно.")
        else:
            sugg = sc.suggest(word)
            msg = f"«{word}» — схоже є помилка.\n\nПідказки:\n" + ("\n".join(sugg) if sugg else "— немає —")
            ans = QMessageBox.warning(self, "Перевірка", msg, QMessageBox.Ok)
            if sugg:
                # Подставить лучший вариант в поле cs
                self.cs.setText(sugg[0])

    def _save(self):
        cs = self.cs.text().strip()
        if not cs:
            QMessageBox.warning(self, "—", "Поле «cs» обов'язкове.")
            return
        uk = self.uk.text().strip()
        ru = self.ru.text().strip()
        pos = self.pos.currentText().strip()
        note = self.note.text().strip()
        if self.word_id:
            update_word(self.db_path, self.word_id,
                        cs=cs, uk=uk, ru=ru, pos=pos, note=note)
        else:
            try:
                self.word_id = add_word(self.db_path, cs=cs, uk=uk, ru=ru, pos=pos, note=note)
            except Exception as e:
                QMessageBox.critical(self, "Помилка", f"Не вдалося зберегти: {e}")
                return
        self.accept()

## Риски и open questions

1. **`pyhunspell` на Windows** — требует нативной библиотеки `hunspell.dll`. Если проблема — fallback на `pyspellchecker` (без словаря, но работает из коробки) или чистый Python вариант на базе `enchant`.
2. **Размер чешского словаря** — `.dic` файл ~1.5 MB, коммитится в репо (норм).
3. **FTS5** — может не быть в очень старых SQLite-сборках Python 3.10+, но в 3.11+ есть по умолчанию.
4. **Unicode в CSV** — используем `utf-8` (не `utf-8-sig`); Excel может не открыть с диакритикой. Опционально добавить `utf-8-sig` для совместимости.
5. **Что делать, если PySide6 не нравится?** — структура (`db.py`, `models.py`, `spellcheck.py`) полностью отделена от Qt, легко заменить UI на Tkinter/Flet/Electron позже.
